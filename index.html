Código backend — Google Apps Script

/*************************************************************
 * CORP. SANTAMARÍA C.A. — Backend del Sistema de Gestión
 * Google Apps Script + Google Sheets
 * 1) Extensiones → Apps Script → pega este archivo
 * 2) Reemplaza HOJA_ID por el ID de tu hoja de cálculo
 * 3) Implementar → Nueva implementación → Aplicación web
 *    (Ejecutar como: Yo | Acceso: Cualquier usuario)
 *************************************************************/

const HOJA_ID = 'PEGA_AQUI_EL_ID_DE_TU_HOJA';
const HOJAS = { Empleados:'Empleados', Pagos:'Pagos', Usuarios:'Usuarios' };

function doGet(e){
  const a = (e.parameter.action || '').toLowerCase();
  try{
    if(a === 'ping')  return ok({ok:true, msg:'Conectado', sistema:'CS Gestion v1'});
    if(a === 'login') return ok(login(e.parameter.usuario, e.parameter.clave));
    if(a === 'list')  return ok(listar(e.parameter.hoja, e.parameter.filtro));
    return ok({ok:false, msg:'Accion no valida'});
  }catch(err){ return ok({ok:false, msg:err.message}); }
}

function doPost(e){
  try{
    const b = JSON.parse(e.postData.contents);
    if(b.action === 'create') return ok(crear(b.hoja, b.data));
    if(b.action === 'update') return ok(actualizar(b.hoja, b.id, b.data));
    if(b.action === 'del')    return ok(eliminar(b.hoja, b.id));
    return ok({ok:false, msg:'Accion no valida'});
  }catch(err){ return ok({ok:false, msg:err.message}); }
}

/* ---------- utilidades ---------- */
function ok(obj){
  return ContentService.createTextOutput(JSON.stringify(obj))
         .setMimeType(ContentService.MimeType.JSON);
}
function sh(nombre){
  const h = SpreadsheetApp.openById(HOJA_ID).getSheetByName(nombre);
  if(!h) throw new Error('No existe la hoja: ' + nombre);
  return h;
}
function cabeceras(h){
  return h.getRange(1,1,1,h.getLastColumn()).getValues()[0].map(v => String(v).trim());
}

/* ---------- operaciones CRUD ---------- */
function listar(nombre, filtro){
  const h = sh(nombre);
  if(h.getLastRow() < 2) return {ok:true, data:[]};
  const vals = h.getDataRange().getValues();
  const head = cabeceras(h);
  let data = vals.slice(1).map(r => { const o={}; head.forEach((c,i)=>o[c]=r[i]); return o; });
  if(filtro){
    const f = String(filtro).toLowerCase();
    data = data.filter(o => JSON.stringify(o).toLowerCase().indexOf(f) > -1);
  }
  return {ok:true, data:data};
}

function crear(nombre, datos){
  const h = sh(nombre), head = cabeceras(h);
  const iId = head.indexOf('ID');
  if(iId > -1 && !datos['ID'])
    datos['ID'] = nombre.slice(0,3).toUpperCase() + '-' + Utilities.getUuid().slice(0,6).toUpperCase();
  if(head.indexOf('Actualizado') > -1) datos['Actualizado'] = new Date();
  h.appendRow(head.map(c => datos[c] !== undefined ? datos[c] : ''));
  return {ok:true, msg:'Registro creado', id:datos['ID']};
}

function actualizar(nombre, id, datos){
  const h = sh(nombre), head = cabeceras(h), iId = head.indexOf('ID');
  const vals = h.getDataRange().getValues();
  for(let i = 1; i < vals.length; i++){
    if(String(vals[i][iId]) === String(id)){
      head.forEach((c,j)=>{ if(datos[c] !== undefined) vals[i][j] = datos[c]; });
      if(head.indexOf('Actualizado') > -1) vals[i][head.indexOf('Actualizado')] = new Date();
      h.getRange(i+1,1,1,head.length).setValues([vals[i]]);
      return {ok:true, msg:'Registro actualizado'};
    }
  }
  return {ok:false, msg:'ID no encontrado: ' + id};
}

function eliminar(nombre, id){
  const h = sh(nombre), vals = h.getDataRange().getValues();
  const iId = cabeceras(h).indexOf('ID');
  for(let i = 1; i < vals.length; i++){
    if(String(vals[i][iId]) === String(id)){
      h.deleteRow(i+1);
      return {ok:true, msg:'Registro eliminado'};
    }
  }
  return {ok:false, msg:'ID no encontrado: ' + id};
}

function login(usuario, clave){
  const r = listar(HOJAS.Usuarios).data.find(u =>
    String(u['Usuario']).toLowerCase() === String(usuario).toLowerCase() &&
    String(u['Clave']) === String(clave) &&
    String(u['Estatus']).toLowerCase() === 'activo'
  );
  if(!r) return {ok:false, msg:'Credenciales invalidas o usuario inactivo'};
  /* Recomendado en produccion: guardar SHA-256 en la columna Clave y comparar:
     const hash = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, clave)
       .map(b => ('0' + ((b + 256) % 256).toString(16)).slice(-2)).join('');  */
  return {ok:true, usuario:r['Usuario'], rol:r['Rol'], nombre:r['Nombre'], cedula:r['Cedula'] || r['Cédula'] || ''};
}
